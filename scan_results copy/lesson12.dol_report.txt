 207 | static void DoState(Core::System& system, PointerWrap& p)
 208 | {
 209 |   bool is_wii = system.IsWii() || system.IsMIOS();
 210 |   const bool is_wii_currently = is_wii;
 211 |   p.Do(is_wii);
 212 |   if (is_wii != is_wii_currently)
 213 |   {
 214 |     OSD::AddMessage(fmt::format("Cannot load a savestate created under {} mode in {} mode",
 215 |                                 is_wii ? "Wii" : "GC", is_wii_currently ? "Wii" : "GC"),
 216 |                     OSD::Duration::NORMAL, OSD::Color::RED);
 217 |     p.SetMeasureMode();
 218 |     return;
 219 |   }
 220 | 
 221 |   // Check to make sure the emulated memory sizes are the same as the savestate
 222 |   auto& memory = system.GetMemory();
 223 |   u32 state_mem1_size = memory.GetRamSizeReal();
 224 |   u32 state_mem2_size = memory.GetExRamSizeReal();
 225 |   p.Do(state_mem1_size);
 226 |   p.Do(state_mem2_size);
 227 |   if (state_mem1_size != memory.GetRamSizeReal() || state_mem2_size != memory.GetExRamSizeReal())
 228 |   {
 229 |     OSD::AddMessage(fmt::format("Memory size mismatch!\n"
 230 |                                 "Current | MEM1 {:08X} ({:3}MB)    MEM2 {:08X} ({:3}MB)\n"
 231 |                                 "State   | MEM1 {:08X} ({:3}MB)    MEM2 {:08X} ({:3}MB)",
 232 |                                 memory.GetRamSizeReal(), memory.GetRamSizeReal() / 0x100000U,
 233 |                                 memory.GetExRamSizeReal(), memory.GetExRamSizeReal() / 0x100000U,
 234 |                                 state_mem1_size, state_mem1_size / 0x100000U, state_mem2_size,
 235 |                                 state_mem2_size / 0x100000U));
 236 |     p.SetMeasureMode();
 237 |     return;
 238 |   }
 239 | 
 240 |   // Movie must be done before the video backend, because the window is redrawn in the video backend
 241 |   // state load, and the frame number must be up-to-date.
 242 |   system.GetMovie().DoState(p);
 243 |   p.DoMarker("Movie");
 244 |   // CoreTiming needs to be restored before restoring Hardware because
 245 |   // the controller code might need to schedule an event if the controller has changed.
 246 |   system.GetCoreTiming().DoState(p);
 247 |   p.DoMarker("CoreTiming");
 248 | 
 249 |   // HW needs to be restored before PowerPC because the data cache might need to be flushed.
 250 |   HW::DoState(system, p);
 251 |   p.DoMarker("HW");
 252 | 
 253 |   g_video_backend->DoState(p);
 254 |   p.DoMarker("video_backend");
 255 | 
 256 |   system.GetPowerPC().DoState(p);
 257 |   p.DoMarker("PowerPC");
 258 | 
 259 |   if (system.IsWii())
 260 |     Wiimote::DoState(p);
 261 |   p.DoMarker("Wiimote");
 262 |   Gecko::DoState(p);
 263 |   p.DoMarker("Gecko");
 264 | 
 265 |   #ifdef USE_RETRO_ACHIEVEMENTS
 266 |     AchievementManager::GetInstance().DoState(p);
 267 |   #endif  // USE_RETRO_ACHIEVEMENTS
 268 | }

 761 | static void CompressBufferToFile(const u8* raw_buffer, u64 size, File::IOFile& f)
 762 | {
 763 |   u64 total_bytes_compressed = 0;
 764 | 
 765 |   while (true)
 766 |   {
 767 |     u64 bytes_left_to_compress = size - total_bytes_compressed;
 768 | 
 769 |     int bytes_to_compress =
 770 |         static_cast<int>(std::min(static_cast<u64>(LZ4_MAX_INPUT_SIZE), bytes_left_to_compress));
 771 |     int compressed_buffer_size = LZ4_compressBound(bytes_to_compress);
 772 |     auto compressed_buffer = std::make_unique<char[]>(compressed_buffer_size);
 773 |     s32 compressed_len =
 774 |         LZ4_compress_default(reinterpret_cast<const char*>(raw_buffer) + total_bytes_compressed,
 775 |                              compressed_buffer.get(), bytes_to_compress, compressed_buffer_size);
 776 | 
 777 |     if (compressed_len == 0)
 778 |     {
 779 |       PanicAlertFmtT("Internal LZ4 Error - compression failed");
 780 |       break;
 781 |     }
 782 | 
 783 |     // The size of the data to write is 'compressed_len'
 784 |     f.WriteArray(&compressed_len, 1);
 785 |     f.WriteBytes(compressed_buffer.get(), compressed_len);
 786 | 
 787 |     total_bytes_compressed += bytes_to_compress;
 788 |     if (total_bytes_compressed == size)
 789 |       break;
 790 |   }
 791 | }

 849 | static void CompressAndDumpState(Core::System& system, CompressAndDumpState_args& save_args)
 850 | {
 851 |   const u8* const buffer_data = save_args.buffer_vector.data();
 852 |   const size_t buffer_size = save_args.buffer_vector.size();
 853 |   const std::string& filename = save_args.filename;
 854 | 
 855 |   // Find free temporary filename.
 856 |   // TODO: The file exists check and the actual opening of the file should be atomic, we don't have
 857 |   // functions for that.
 858 |   std::string temp_filename;
 859 |   size_t temp_counter = static_cast<size_t>(Common::CurrentThreadId());
 860 |   do
 861 |   {
 862 |     temp_filename = fmt::format("{}{}.tmp", filename, temp_counter);
 863 |     ++temp_counter;
 864 |   } while (File::Exists(temp_filename));
 865 | 
 866 |   File::IOFile f(temp_filename, "wb");
 867 |   if (!f)
 868 |   {
 869 |     Core::DisplayMessage("Failed to create state file", 2000);
 870 |     return;
 871 |   }
 872 | 
 873 |   WriteHeadersToFile(buffer_size, f);
 874 | 
 875 |   if (s_use_compression)
 876 |     CompressBufferToFile(buffer_data, buffer_size, f);
 877 |   else
 878 |     f.WriteBytes(buffer_data, buffer_size);
 879 | 
 880 |   if (!f.IsGood())
 881 |     Core::DisplayMessage("Failed to write state file", 2000);
 882 | 
 883 |   const std::string last_state_filename = File::GetUserPath(D_STATESAVES_IDX) + "lastState.sav";
 884 |   const std::string last_state_dtmname = last_state_filename + ".dtm";
 885 |   const std::string dtmname = filename + ".dtm";
 886 | 
 887 |   {
 888 |     std::lock_guard lk(s_save_thread_mutex);
 889 | 
 890 |     // Backup existing state (overwriting an existing backup, if any).
 891 |     if (File::Exists(filename))
 892 |     {
 893 |       if (File::Exists(last_state_filename))
 894 |         File::Delete((last_state_filename));
 895 |       if (File::Exists(last_state_dtmname))
 896 |         File::Delete((last_state_dtmname));
 897 | 
 898 |       if (!File::Rename(filename, last_state_filename))
 899 |       {
 900 |         Core::DisplayMessage("Failed to move previous state to state undo backup", 1000);
 901 |       }
 902 |       else if (File::Exists(dtmname))
 903 |       {
 904 |         if (!File::Rename(dtmname, last_state_dtmname))

 906 |       }
 907 |     }

 909 |     auto& movie = system.GetMovie();
 910 |     if ((movie.IsMovieActive()) && !movie.IsJustStartingRecordingInputFromSaveState())

 912 |     else if (!movie.IsMovieActive())

 923 |       Core::DisplayMessage("Failed to rename state file", 2000);

1346 | static bool DecompressLZ4(std::vector<u8>& raw_buffer, u64 size, File::IOFile& f)
1347 | {
1348 |   raw_buffer.resize(size);
1349 | 
1350 |   u64 total_bytes_read = 0;
1351 |   while (true)
1352 |   {
1353 |     s32 compressed_data_len;
1354 |     if (!f.ReadArray(&compressed_data_len, 1))
1355 |     {
1356 |       PanicAlertFmt("Could not read state data length");
1357 |       return false;
1358 |     }
1359 | 
1360 |     if (compressed_data_len <= 0)
1361 |     {
1362 |       PanicAlertFmtT("Internal LZ4 Error - Tried decompressing {0} bytes", compressed_data_len);
1363 |       return false;
1364 |     }
1365 | 
1366 |     auto compressed_data = std::make_unique<char[]>(compressed_data_len);
1367 |     if (!f.ReadBytes(compressed_data.get(), compressed_data_len))
1368 |     {
1369 |       PanicAlertFmt("Could not read state data");
1370 |       return false;
1371 |     }
1372 | 
1373 |     u32 max_decompress_size =
1374 |         static_cast<u32>(std::min((u64)LZ4_MAX_INPUT_SIZE, size - total_bytes_read));
1375 | 
1376 |     int bytes_read = LZ4_decompress_safe(
1377 |         compressed_data.get(), reinterpret_cast<char*>(raw_buffer.data()) + total_bytes_read,
1378 |         compressed_data_len, max_decompress_size);
1379 | 
1380 |     if (bytes_read < 0)
1381 |     {
1382 |       PanicAlertFmtT("Internal LZ4 Error - decompression failed ({0}, {1}, {2})", bytes_read,
1383 |                      compressed_data_len, max_decompress_size);
1384 |       return false;
1385 |     }
1386 | 
1387 |     total_bytes_read += static_cast<u64>(bytes_read);
1388 | 
1389 |     if (total_bytes_read == size)
1390 |     {
1391 |       return true;
1392 |     }
1393 |     else if (total_bytes_read > size)
1394 |     {
1395 |       PanicAlertFmtT("Internal LZ4 Error - payload size mismatch ({0} / {1}))", total_bytes_read,
1396 |                      size);
1397 |       return false;
1398 |     }
1399 |   }
